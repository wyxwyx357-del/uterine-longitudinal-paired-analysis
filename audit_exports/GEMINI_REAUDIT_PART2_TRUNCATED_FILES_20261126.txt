def read_csv(path: Path) -> tuple[list[str], list[dict[str, str]]]:
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        return list(reader.fieldnames or []), list(reader)


def write_csv(path: Path, fields: list[str], rows: list[dict[str, str]]) -> None:
    with path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, extrasaction="ignore")
        writer.writeheader()
        writer.writerows(rows)


def complete_output(path: Path) -> bool:

        len(rows) == 1

    temporary = path.with_suffix(".tmp")

    parser.add_argument("--report", type=Path, required=True)

    with args.manifest.open("r", encoding="utf-8-sig", newline="") as stream:

        source_rows = list(reader)
    if not source_rows:
        raise ValueError("Manifest contains no patients")

    seen: set[str] = set()
    report_rows: list[dict[str, object]] = []
    for row in source_rows:
        case_id = row["case_id"]
        problems: list[str] = []
        if case_id in seen:
            problems.append("duplicate_case_id")
        seen.add(case_id)

        paths = {key: Path(row[key]) for key in ("video", "tracking", "nodes")}
        for key, path in paths.items():
            if not path.is_absolute():
                problems.append(f"{key}_not_absolute")
            if not path.is_file():
                problems.append(f"{key}_missing")

        frame_count = ""
        fps = ""
        width = ""
        height = ""
        if paths["video"].is_file():
            capture = cv2.VideoCapture(str(paths["video"]))
            try:
                if not capture.isOpened():
                    problems.append("video_open_failed")
                else:
                    frame_count = int(capture.get(cv2.CAP_PROP_FRAME_COUNT))
                    fps = capture.get(cv2.CAP_PROP_FPS)
                    width = int(capture.get(cv2.CAP_PROP_FRAME_WIDTH))
                    height = int(capture.get(cv2.CAP_PROP_FRAME_HEIGHT))
                    if (width, height) != (1260, 910):
                        problems.append(
                            f"unsupported_video_geometry:{width}x{height};"
                            "expected_1260x910"
                        )
                    ok, frame = capture.read()
                    if not ok or frame is None:
                        problems.append("video_first_frame_read_failed")
            finally:
                capture.release()

        tracking_keys = ""
        if paths["tracking"].is_file():
            try:
                with np.load(paths["tracking"], allow_pickle=False) as tracking:
                    tracking_keys = "|".join(sorted(tracking.files))
            except Exception as exc:  # report the input issue without altering it
                problems.append(f"tracking_load_failed:{type(exc).__name__}")

        if paths["nodes"].is_file():
            try:
                with paths["nodes"].open("r", encoding="utf-8-sig") as stream:
                    json.load(stream)
            except Exception as exc:  # report the input issue without altering it
                problems.append(f"nodes_load_failed:{type(exc).__name__}")

        report_rows.append({
            "case_id": case_id,
            "status": "ready" if not problems else ";".join(problems),
            "video": str(paths["video"]),
            "frame_count": frame_count,
            "fps": fps,
            "width": width,
            "height": height,
            "tracking": str(paths["tracking"]),
            "tracking_keys": tracking_keys,
            "nodes": str(paths["nodes"]),
        })

    print(f"validation_failed_cases={len(report_rows) - ready}")

"""Outcome-blind longitudinal (cervix-to-fundus) tissue-motion frequency.

Reuses background-corrected trajectory caches produced by the tracking stage and the

import json
import math
import os
import shutil

GRID_CPM = np.round(np.arange(0.5, 5.0001, 0.05), 8)
EVIDENCE_THRESHOLDS = {
    "control_fixed_frequency_score": 0.50,
    "control_periodic_correlation": 0.70,
    "background_to_tissue_range_ratio": 0.50,
    "supported_point_phase_coherence": 0.70,

    "minimum_mean_ncc": 0.80,

}


def safe_correlation(a: np.ndarray, b: np.ndarray) -> float:
    valid = np.isfinite(a) & np.isfinite(b)
    if valid.sum() < 20 or np.std(a[valid]) < 1e-12 or np.std(b[valid]) < 1e-12:
        return math.nan
    return float(np.corrcoef(a[valid], b[valid])[0, 1])


def fit_unified_frequency(time_s: np.ndarray, signal: np.ndarray) -> dict:
    if len(time_s) < 20:
        return {"status": "missing", "reason": "technical_segment_too_short"}
    tt = np.asarray(time_s, float) - float(np.mean(time_s))
    signal = np.asarray(signal, float)
    trend_x = np.column_stack((np.ones(len(tt)), tt))
    trend = trend_x @ np.linalg.lstsq(trend_x, signal, rcond=None)[0]
    residual = signal - trend
    energy = float(residual @ residual)
    if energy / len(signal) < 1e-12:
        return {"status": "missing", "reason": "technical_nearly_constant"}
    scores = []
    for frequency in GRID_CPM:
        phase = 2.0 * np.pi * tt * frequency / 60.0
        design = np.column_stack((trend_x, np.sin(phase), np.cos(phase)))
        fitted = design @ np.linalg.lstsq(design, signal, rcond=None)[0]
        scores.append(np.clip(1.0 - float(((signal - fitted) ** 2).sum()) / energy, 0.0, 1.0))
    scores = np.asarray(scores)
    index = int(np.argmax(scores))
    frequency = float(GRID_CPM[index])
    return {
        "status": "valid",
        "reason": "unified_fixed_band_dominant_frequency",
        "dominant_frequency_cpm": frequency,
        "period_s": 60.0 / frequency,
        "score": float(scores[index]),
        "search_boundary": bool(index in (0, len(GRID_CPM) - 1)),
        "observed_cycles": float(frequency * (time_s[-1] - time_s[0]) / 60.0),
    }


def periodic_trend_comparison(time_s: np.ndarray, signal: np.ndarray, frequency: float) -> dict:
    tt = np.asarray(time_s, float) - float(np.mean(time_s))
    y = np.asarray(signal, float)
    quadratic = np.column_stack((np.ones(len(tt)), tt, tt * tt))
    phase = 2.0 * np.pi * tt * frequency / 60.0
    periodic = np.column_stack((np.ones(len(tt)), tt, np.sin(phase), np.cos(phase)))
    q_residual = y - quadratic @ np.linalg.lstsq(quadratic, y, rcond=None)[0]
    p_residual = y - periodic @ np.linalg.lstsq(periodic, y, rcond=None)[0]
    q_sse = float(q_residual @ q_residual)
    p_sse = float(p_residual @ p_residual)
    return {
        "quadratic_trend_sse": q_sse,
        "periodic_model_sse": p_sse,
        "periodic_over_quadratic_improvement": (
            (q_sse - p_sse) / q_sse if q_sse > 1e-12 else math.nan
        ),
    }


def fit_candidate(time_s: np.ndarray, signal: np.ndarray) -> dict:
    fit = fit_unified_frequency(time_s, signal)
    if fit.get("status") != "valid":
        return fit
    return {
        **fit,
        **periodic_trend_comparison(time_s, signal, float(fit["dominant_frequency_cpm"])),
    }


def usability(fit: dict, duration_s: float) -> tuple[bool, str]:
    reasons = []
    if fit.get("status") != "valid":
        reasons.append(str(fit.get("reason", "candidate_frequency_missing")))
    else:
        if duration_s + 1e-12 < 15.0:
            reasons.append("continuous_duration_below_rule")
        if float(fit.get("observed_cycles", 0.0)) + 1e-12 < 1.0:
            reasons.append("candidate_cycle_coverage_below_rule")
        improvement = fit.get("periodic_over_quadratic_improvement", math.nan)
        if not np.isfinite(improvement) or improvement < 0.05:
            reasons.append("periodic_model_not_better_than_quadratic_trend")
    return not reasons, "pass" if not reasons else ";".join(reasons)


def fixed_frequency_component(time_s: np.ndarray, signal: np.ndarray, frequency: float) -> dict:
    time_s = np.asarray(time_s, float)
    signal = np.asarray(signal, float)
    valid = np.isfinite(time_s) & np.isfinite(signal)
    if valid.sum() < 20:
        return {"score": math.nan, "amplitude": math.nan}
    tt = time_s[valid] - float(np.mean(time_s[valid]))
    y = signal[valid]
    trend = np.column_stack((np.ones(len(tt)), tt))
    trend_fitted = trend @ np.linalg.lstsq(trend, y, rcond=None)[0]
    residual_valid = y - trend_fitted
    energy = float(residual_valid @ residual_valid)
    if energy <= 1e-12:
        return {"score": math.nan, "amplitude": 0.0}
    phase = 2.0 * np.pi * tt * frequency / 60.0
    waves = np.column_stack((np.sin(phase), np.cos(phase)))
    design = np.column_stack((trend, waves))
    coefficients = np.linalg.lstsq(design, y, rcond=None)[0]
    periodic_valid = waves @ coefficients[-2:]
    full_residual = y - design @ coefficients
    residual = np.full(len(signal), np.nan)
    periodic = np.full(len(signal), np.nan)
    residual[valid] = residual_valid
    periodic[valid] = periodic_valid
    return {
        "score": float(np.clip(1.0 - (full_residual @ full_residual) / energy, 0, 1)),
        "amplitude": float(np.linalg.norm(coefficients[-2:])),
        "residual": residual,
        "periodic": periodic,
        "coefficient": complex(float(coefficients[-2]), float(coefficients[-1])),
    }

    tracked: dict, initial: np.ndarray, axes: np.ndarray,
    columns: np.ndarray, start: int, stop: int,
) -> np.ndarray:
    values = np.full(stop - start, np.nan)
    for out_index, frame in enumerate(range(start, stop)):
        if not tracked["pose_valid"][frame]:
            continue
        inverse = np.linalg.inv(tracked["poses"][frame])
        mapped = initial[columns] @ inverse[:2, :2].T + inverse[:2, 2]
        values[out_index] = np.einsum(
            "ki,ki->k", mapped - initial[columns], axes[columns]
        ).mean()
    return values


def artifact_evidence(
    tracked: dict, initial: np.ndarray, axes: np.ndarray,
    selected: dict, fit: dict, usable: bool,
) -> dict:
    start, stop = selected["start"], selected["stop"]
    columns = selected["columns"]
    time_s = tracked["time_s"][start:stop]
    frequency = finite_float(fit.get("dominant_frequency_cpm"))
    displacement = tracked["corrected"][start:stop, columns] - initial[columns]
    point_signals = np.einsum("tki,ki->tk", displacement, axes[columns])
    tissue = point_signals.mean(axis=1)
    background = background_signal(tracked, initial, axes, columns, start, stop)
    local_brightness = np.nanmean(tracked["point_brightness"][start:stop, columns], axis=1)
    global_brightness = tracked["global_brightness"][start:stop]
    tissue_component = fixed_frequency_component(time_s, tissue, frequency)
    background_component = fixed_frequency_component(time_s, background, frequency)
    local_component = fixed_frequency_component(time_s, local_brightness, frequency)
    global_component = fixed_frequency_component(time_s, global_brightness, frequency)
    point_components = [
        fixed_frequency_component(time_s, point_signals[:, index], frequency)
        for index in range(point_signals.shape[1])
    ]
    coefficients = np.asarray([
        item.get("coefficient", 0j)
        if np.isfinite(abs(item.get("coefficient", complex(math.nan, math.nan)))) else 0j
        for item in point_components
    ])
    amplitudes = np.abs(coefficients)
    amp_sum = float(amplitudes.sum())
    amp_sq_sum = float(amplitudes @ amplitudes)
    participation = amp_sum ** 2 / (len(amplitudes) * amp_sq_sum) if amp_sq_sum > 1e-24 else 0.0
    coherence = float(abs(coefficients.sum()) / amp_sum) if amp_sum > 1e-12 else math.nan

    def periodic_corr(control: dict) -> float:
        if "periodic" not in tissue_component or "periodic" not in control:
            return math.nan
        return safe_correlation(tissue_component["periodic"], control["periodic"])

    background_corr = periodic_corr(background_component)
    local_corr = periodic_corr(local_component)
    global_corr = periodic_corr(global_component)
    tissue_range = float(np.ptp(tissue))
    finite_background = background[np.isfinite(background)]
    background_range = float(np.ptp(finite_background)) if len(finite_background) else math.nan
    background_ratio = background_range / tissue_range if tissue_range > 1e-12 else math.nan
    threshold = EVIDENCE_THRESHOLDS
    background_flag = bool(
        np.isfinite(background_ratio)
        and background_ratio >= threshold["background_to_tissue_range_ratio"]
        and background_component.get("score", math.nan) >= threshold["control_fixed_frequency_score"]
        and abs(background_corr) >= threshold["control_periodic_correlation"]
    )
    brightness_flag = bool(
        global_component.get("score", math.nan) >= threshold["control_fixed_frequency_score"]
        and abs(global_corr) >= threshold["control_periodic_correlation"]
    )
    ncc = tracked["target_ncc"][start:stop, columns]
    fb = tracked["target_fb"][start:stop, columns]
    mean_ncc = float(np.nanmean(ncc)) if np.isfinite(ncc).any() else math.nan
    p90_fb = float(np.nanpercentile(fb, 90)) if np.isfinite(fb).any() else math.nan
    heldout = tracked["heldout_corrected"][start:stop]
    mean_heldout = float(np.nanmean(heldout)) if np.isfinite(heldout).any() else math.nan
    tracking_flag = bool(
        participation < threshold["minimum_point_participation_fraction"]
        or (np.isfinite(coherence) and coherence < threshold["supported_point_phase_coherence"])
        or (np.isfinite(mean_ncc) and mean_ncc < threshold["minimum_mean_ncc"])
        or (np.isfinite(p90_fb) and p90_fb > threshold["maximum_p90_forward_backward_px"])
    )
    coordinated = bool(
        usable and not tracking_flag
        and participation >= threshold["minimum_point_participation_fraction"]
        and np.isfinite(coherence)
        and coherence >= threshold["supported_point_phase_coherence"]
    )
    if tracking_flag:
        evidence_class, reason = "artifact_risk", "tracking_or_point_inconsistency"
    elif coordinated:
        warnings = []
        if background_flag:
            warnings.append("background_same_frequency_warning")
        if brightness_flag:
            warnings.append("global_brightness_same_frequency_warning")
        reason = "usable_frequency;participating_coherent_points"
        if warnings:
            reason += ";" + ";".join(warnings)
        evidence_class = "coordinated_tissue_displacement_supported"
    else:
        evidence_class, reason = "insufficient_evidence", "longitudinal_motion_support_incomplete"
    return {
        "evidence_class": evidence_class,
        "evidence_reason": reason,
        "background_same_frequency_flag": background_flag,
        "brightness_same_frequency_flag": brightness_flag,
        "tracking_artifact_flag": tracking_flag,
        "same_frequency_confounding_flag": bool(background_flag or brightness_flag),
        "tissue_fixed_frequency_score": tissue_component.get("score", ""),
        "background_fixed_frequency_score": background_component.get("score", ""),
        "background_periodic_correlation": background_corr,
        "background_to_tissue_range_ratio_same_segment": background_ratio,
        "local_brightness_fixed_frequency_score": local_component.get("score", ""),
        "local_brightness_periodic_correlation": local_corr,
        "global_brightness_fixed_frequency_score": global_component.get("score", ""),
        "global_brightness_periodic_correlation": global_corr,
        "point_phase_coherence": coherence,
        "point_participation_fraction": participation,
        "mean_target_ncc_same_segment": mean_ncc,
        "p90_target_fb_px_same_segment": p90_fb,
        "mean_background_heldout_residual_px_same_segment": mean_heldout,
    }


def local_tangents(points: np.ndarray) -> np.ndarray:
    """Fixed first-frame local tangents, oriented by stored cervix->fundus order."""
    points = np.asarray(points, float)
    if len(points) < 3 or points.shape[1] != 2:
        raise ValueError("Each wall needs at least three ordered 2-D points")
    delta = np.empty_like(points)
    delta[0] = points[1] - points[0]
    delta[-1] = points[-1] - points[-2]
    delta[1:-1] = points[2:] - points[:-2]
    norm = np.linalg.norm(delta, axis=1)
    if not np.isfinite(norm).all() or np.any(norm < 1e-8):
        raise ValueError("Degenerate longitudinal tangent")
    return delta / norm[:, None]


def parse_choice(value: object) -> tuple[int, ...]:
    if pd.isna(value) or str(value).strip() == "":
        return ()
    return tuple(int(item) for item in str(value).split(";") if item != "")


def finite_float(value: object, default: float = math.nan) -> float:
    try:
        result = float(value)
    except (TypeError, ValueError):
        return default
    return result if math.isfinite(result) else default


def synthetic_checks() -> dict[str, object]:
    time_s = np.arange(0.0, 80.0, 1.0 / 25.0)
    points = np.array([[0.0, 0.0], [1.0, 0.2], [2.0, 0.3]], float)
    tangent = local_tangents(points)
    recovered = {}
    for frequency in (1.0, 2.0, 3.0):
        wave = 1.2 * np.sin(2 * np.pi * frequency * time_s / 60.0 + 0.3)
        positions = points[None] + wave[:, None, None] * tangent[None]
        signal = np.einsum(
            "tki,ki->tk", positions - points[None], tangent
        ).mean(axis=1)
        fit = fit_candidate(time_s, signal)
        estimate = finite_float(fit.get("dominant_frequency_cpm"))
        recovered[str(frequency)] = estimate
        if not math.isfinite(estimate) or abs(estimate - frequency) > 0.05 + 1e-12:
            raise AssertionError(f"Known longitudinal frequency failed: {frequency}->{estimate}")

    normal = np.column_stack((-tangent[:, 1], tangent[:, 0]))
    cross_wave = np.sin(2 * np.pi * 2.0 * time_s / 60.0)
    cross_positions = points[None] + cross_wave[:, None, None] * normal[None]
    cross_signal = np.einsum(
        "tki,ki->tk", cross_positions - points[None], tangent
    ).mean(axis=1)
    cross_leakage = float(np.max(np.abs(cross_signal)))
    if cross_leakage > 1e-10:
        raise AssertionError(f"Cross-wall motion leaked into longitudinal signal: {cross_leakage}")

    amplitudes = np.array([1.0, 0.0, 0.0])
    participation = float(
        amplitudes.sum() ** 2 / (len(amplitudes) * (amplitudes @ amplitudes))
    )
    if not participation < 2.0 / 3.0:
        raise AssertionError("Single-point participation check failed")

    # Pure rigid background motion must disappear after perfect correction.
    background_wave = 2.0 * np.sin(2 * np.pi * 2.0 * time_s / 60.0)
    pure_background_corrected = np.repeat(points[None], len(time_s), axis=0)
    pure_background_signal = np.einsum(
        "tki,ki->tk", pure_background_corrected - points[None], tangent
    ).mean(axis=1)
    pure_background_fit = fit_candidate(time_s, pure_background_signal)
    pure_background_usable, _ = usability(
        pure_background_fit, float(time_s[-1] - time_s[0])
    )
    if pure_background_usable:
        raise AssertionError("Perfectly corrected pure background motion was retained")

    # True local motion may coexist at exactly the same frequency as background.
    # Same-frequency background is a warning, not proof that corrected motion is artifact.
    local_wave = 1.2 * np.sin(2 * np.pi * 2.0 * time_s / 60.0 + 0.3)
    local_corrected = points[None] + local_wave[:, None, None] * tangent[None]
    poses = np.repeat(np.eye(3)[None], len(time_s), axis=0)
    poses[:, 0, 2] = -background_wave
    tracked = {
        "time_s": time_s,
        "corrected": local_corrected,
        "valid": np.ones((len(time_s), len(points)), dtype=bool),
        "pose_valid": np.ones(len(time_s), dtype=bool),
        "poses": poses,
        "target_ncc": np.ones((len(time_s), len(points))),
        "target_fb": np.zeros((len(time_s), len(points))),
        "point_brightness": np.ones((len(time_s), len(points))),
        "global_brightness": np.ones(len(time_s)),
        "heldout_corrected": np.zeros(len(time_s)),
    }
    same_frequency_fit = fit_candidate(time_s, local_wave)
    same_frequency_usable, _ = usability(
        same_frequency_fit, float(time_s[-1] - time_s[0])
    )
    same_frequency_evidence = artifact_evidence(
        tracked,
        points,
        tangent,
        {"start": 0, "stop": len(time_s), "columns": np.arange(len(points))},
        same_frequency_fit,
        same_frequency_usable,
    )
    if not same_frequency_evidence["background_same_frequency_flag"]:
        raise AssertionError("Same-frequency background warning was not detected")
    if same_frequency_evidence["evidence_class"] != "coordinated_tissue_displacement_supported":
        raise AssertionError("True local motion was rejected only because background had the same frequency")
    same_frequency_region, _ = analyse_region(
        pd.Series({
            "case_id": "synthetic_same_frequency",
            "wall": "anterior",
            "region": "middle",
            "selected_local_point_indices": "0;1;2",
            "selected_start_frame": 0,
            "selected_stop_frame_exclusive": len(time_s),
            "selected_point_count": len(points),
        }),
        tracked,
        points,
        tangent,
        tangent,
        np.arange(len(points)),
        np.arange(len(points)),
    )
    if not same_frequency_region["retained_for_longitudinal_model"]:
        raise AssertionError("Same-frequency true local motion was not retained")
    if same_frequency_region["longitudinal_technical_confidence"] != "medium":
        raise AssertionError("Same-frequency warning did not downgrade confidence to medium")

    brightness_tracked = dict(tracked)
    brightness_tracked["poses"] = np.repeat(np.eye(3)[None], len(time_s), axis=0)
    brightness_tracked["global_brightness"] = 100.0 + 5.0 * local_wave
    brightness_evidence = artifact_evidence(
        brightness_tracked,
        points,
        tangent,
        {"start": 0, "stop": len(time_s), "columns": np.arange(len(points))},
        same_frequency_fit,
        same_frequency_usable,
    )
    if not brightness_evidence["brightness_same_frequency_flag"]:
        raise AssertionError("Same-frequency brightness warning was not detected")
    if brightness_evidence["evidence_class"] != "coordinated_tissue_displacement_supported":
        raise AssertionError("True local motion was rejected only because brightness had the same frequency")
    return {
        "status": "pass",
        "known_frequency_recovery_cpm": recovered,
        "pure_cross_wall_projection_leakage_px": cross_leakage,
        "pure_background_after_perfect_correction_retained": pure_background_usable,
        "same_frequency_background_warning_detected": bool(
            same_frequency_evidence["background_same_frequency_flag"]
        ),
        "true_local_motion_with_same_frequency_background_class": (
            same_frequency_evidence["evidence_class"]
        ),
        "true_local_motion_with_same_frequency_background_confidence": (
            same_frequency_region["longitudinal_technical_confidence"]
        ),
        "same_frequency_brightness_warning_detected": bool(
            brightness_evidence["brightness_same_frequency_flag"]
        ),
        "true_local_motion_with_same_frequency_brightness_class": (
            brightness_evidence["evidence_class"]
        ),
        "single_moving_point_participation_fraction": participation,
    }


def cache_map(run_root: Path) -> dict[str, Path]:
    """Return caches only for cases that completed the tracking stage."""
    completed: set[str] | None = None
    status_path = run_root / "resumable_status.json"
    if status_path.is_file():
        status = json.loads(status_path.read_text(encoding="utf-8"))
        completed = {
            str(case_id)
            for case_id, record in status.get("cases", {}).items()
            if record.get("state") == "complete"
        }

    mapping: dict[str, Path] = {}
    for path in run_root.glob("cases/case_*/trajectory_cache/*.npz"):
        with np.load(path, allow_pickle=False) as archive:
            case_id = str(archive["case_id"][0]).strip()
        if completed is not None and case_id not in completed:
            continue
        if case_id in mapping:
            raise ValueError(f"Duplicate trajectory cache for {case_id}")
        mapping[case_id] = path
    return mapping


def selected_segment(row: pd.Series, columns: np.ndarray) -> dict | None:
    choice = parse_choice(row.get("selected_local_point_indices"))
    start = int(finite_float(row.get("selected_start_frame"), -1))
    stop = int(finite_float(row.get("selected_stop_frame_exclusive"), -1))
    if not choice or start < 0 or stop <= start:
        return None
    if max(choice) >= len(columns):
        raise ValueError("Saved local point choice is outside its region")
    return {
        "start": start,
        "stop": stop,
        "columns": columns[list(choice)],
        "local_choice": choice,
        "duration_s": finite_float(row.get("optimized_segment_duration_s"), 0.0),
    }


def analyse_region(
    row: pd.Series,
    tracked: dict,
    initial: np.ndarray,
    longitudinal_axes: np.ndarray,
    inward_axes: np.ndarray,
    local_indices: np.ndarray,
    columns: np.ndarray,
) -> tuple[dict[str, object], dict[str, np.ndarray] | None]:
    selected = selected_segment(row, columns)
    base = {
        "case_id": row["case_id"],
        "wall": row["wall"],
        "region": row["region"],
        "source_selected_point_count": int(finite_float(row.get("selected_point_count"), 0)),
        "source_selected_point_fraction": row.get("selected_point_fraction", ""),
        "source_selected_span_fraction": row.get("selected_span_fraction", ""),
        "source_selected_local_point_indices": row.get("selected_local_point_indices", ""),
        "source_segment_start_frame": row.get("selected_start_frame", ""),
        "source_segment_stop_frame_exclusive": row.get("selected_stop_frame_exclusive", ""),
        "source_inward_frequency_cpm": row.get("optimized_candidate_frequency_cpm", ""),
        "source_inward_technical_confidence": row.get("technical_confidence", ""),
    }
    if selected is None:
        return {
            **base,
            "longitudinal_candidate_status": "missing",
            "longitudinal_candidate_reason": "source_fixed_point_segment_missing",
            "frequency_usable": False,
            "longitudinal_technical_confidence": "excluded",
            "retained_for_longitudinal_model": False,
        }, None

    start, stop = selected["start"], selected["stop"]
    chosen = selected["columns"]
    if stop > len(tracked["time_s"]):
        raise ValueError("Saved segment exceeds trajectory cache")
    time_s = tracked["time_s"][start:stop]
    displacement = tracked["corrected"][start:stop, chosen] - initial[chosen]
    point_signals = np.einsum(
        "tki,ki->tk", displacement, longitudinal_axes[chosen]
    )
    signal = point_signals.mean(axis=1)
    fit = fit_candidate(time_s, signal)
    duration_s = float(time_s[-1] - time_s[0]) if len(time_s) > 1 else 0.0
    usable, usability_reason = usability(fit, duration_s)
    evidence = artifact_evidence(
        tracked, initial, longitudinal_axes, selected, fit, usable
    )
    coordinated = evidence.get("evidence_class") == "coordinated_tissue_displacement_supported"
    retained = bool(usable and coordinated)
    cycles = finite_float(fit.get("observed_cycles"), 0.0)
    same_frequency_warning = bool(evidence.get("same_frequency_confounding_flag", False))
    confidence = (
        "high" if retained and cycles >= 2.0 and not same_frequency_warning
        else "medium" if retained
        else "excluded"
    )

    component = {}
    if fit.get("status") == "valid":
        component = fixed_frequency_component(
            time_s, signal, float(fit["dominant_frequency_cpm"])
        )
    coefficient = component.get("coefficient", complex(math.nan, math.nan))
    phase = (
        float(math.atan2(coefficient.imag, coefficient.real))
        if math.isfinite(coefficient.real) and math.isfinite(coefficient.imag)
        else math.nan
    )
    dot = np.sum(
        longitudinal_axes[chosen] * inward_axes[chosen], axis=1
    )
    angle = np.degrees(np.arccos(np.clip(np.abs(dot), 0.0, 1.0)))
    effective_min = 60.0 / duration_s if duration_s > 0 else math.nan

    output = {
        **base,
        "longitudinal_candidate_status": fit.get("status", "missing"),
        "longitudinal_candidate_reason": fit.get("reason", ""),
        "longitudinal_frequency_cpm": fit.get("dominant_frequency_cpm", ""),
        "longitudinal_period_s": fit.get("period_s", ""),
        "longitudinal_score": fit.get("score", ""),
        "longitudinal_amplitude_px": component.get("amplitude", ""),
        "longitudinal_phase_rad": phase if math.isfinite(phase) else "",
        "longitudinal_displacement_range_px": float(np.ptp(signal)),
        "segment_duration_s": duration_s,
        "observed_cycles": fit.get("observed_cycles", ""),
        "effective_minimum_one_cycle_frequency_cpm": effective_min,
        "search_boundary": fit.get("search_boundary", ""),
        "periodic_over_quadratic_improvement": fit.get(
            "periodic_over_quadratic_improvement", ""
        ),
        "frequency_usable": bool(usable),
        "frequency_usability_reason": usability_reason,
        "evidence_class": (
            "longitudinal_coordinated_motion_supported" if coordinated
            else evidence.get("evidence_class", "insufficient_evidence")
        ),
        "evidence_reason": evidence.get("evidence_reason", ""),
        "longitudinal_technical_confidence": confidence,
        "retained_for_longitudinal_model": retained,
        "cycle_reliability": (
            "two_or_more_cycles" if cycles >= 2.0 else
            "one_to_less_than_two_cycles" if cycles >= 1.0 else
            "below_one_cycle"
        ),
        "mean_absolute_tangent_inward_dot": float(np.mean(np.abs(dot))),
        "median_tangent_inward_angle_deg": float(np.median(angle)),
        **{key: value for key, value in evidence.items() if key not in {"evidence_class", "evidence_reason"}},
    }
    diagnostic = None
    if component and "residual" in component and "periodic" in component:
        diagnostic = {
            "time_s": time_s - time_s[0],
            "residual": component["residual"],
            "periodic": component["periodic"],
        }
    return output, diagnostic


def patient_summary(rows: list[dict[str, object]]) -> pd.DataFrame:
    grouped: dict[str, list[dict[str, object]]] = defaultdict(list)
    for row in rows:
        grouped[str(row["case_id"])].append(row)
    output = []
    for case_id, own in grouped.items():
        retained = [row for row in own if row.get("retained_for_longitudinal_model")]
        output.append({
            "case_id": case_id,
            "regions_total": len(own),
            "retained_longitudinal_region_count": len(retained),
            "retained_anterior_count": sum(row["wall"] == "anterior" for row in retained),
            "retained_posterior_count": sum(row["wall"] == "posterior" for row in retained),
            "high_confidence_count": sum(row.get("longitudinal_technical_confidence") == "high" for row in own),
            "medium_confidence_count": sum(row.get("longitudinal_technical_confidence") == "medium" for row in own),
            "two_or_more_cycles_count": sum(row.get("cycle_reliability") == "two_or_more_cycles" for row in retained),
        })
    return pd.DataFrame(output).sort_values("case_id")


def plot_diagnostics(
    output_dir: Path,
    selected_cases: list[str],
    diagnostics: dict[tuple[str, str, str], dict[str, np.ndarray]],
    rows: list[dict[str, object]],
) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)
    lookup = {(str(row["case_id"]), str(row["wall"]), str(row["region"])): row for row in rows}
    for case_id in selected_cases:
        figure, axes = plt.subplots(2, 3, figsize=(15, 7), constrained_layout=True)
        for axis, wall, region in zip(
            axes.ravel(),
            ["anterior"] * 3 + ["posterior"] * 3,
            list(REGION_NAMES) * 2,
        ):
            key = (case_id, wall, region)
            diagnostic = diagnostics.get(key)
            row = lookup.get(key, {})
            if diagnostic is None:
                axis.text(0.5, 0.5, "no retained curve", ha="center", va="center")
            else:
                axis.plot(diagnostic["time_s"], diagnostic["residual"], lw=1.0, label="detrended longitudinal displacement")
                axis.plot(diagnostic["time_s"], diagnostic["periodic"], lw=2.0, label="fitted periodic component")
            axis.set_title(
                f"{wall}-{region} | f={row.get('longitudinal_frequency_cpm', '')} cpm | {row.get('longitudinal_technical_confidence', '')}"
            )
            axis.set_xlabel("s")
            axis.set_ylabel("px")
            axis.grid(alpha=0.25)
        axes[0, 0].legend(fontsize=8)
        figure.suptitle(f"Longitudinal cervix-to-fundus motion: {case_id}")
        safe = "".join(c if c.isalnum() or c in "-_" else "_" for c in case_id)
        figure.savefig(output_dir / f"{safe}.png", dpi=150)
        plt.close(figure)


def main() -> None:
    parser = argparse.ArgumentParser(description="Longitudinal tissue-motion frequency from frozen R2 caches")
    parser.add_argument("--source-run", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    source_run = args.source_run.resolve()
    output = args.output.resolve()
    if output.exists():
        raise FileExistsError(f"Refusing to overwrite: {output}")
    output.mkdir(parents=True)

    checks = synthetic_checks()
    (output / "synthetic_checks.json").write_text(
        json.dumps(checks, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    region_file = source_run / "merged_results" / "optimized_region_frequency.csv"
    source_regions = pd.read_csv(region_file, dtype={"case_id": str})
    source_regions["case_id"] = source_regions["case_id"].astype(str).str.strip()
    caches = cache_map(source_run)
    expected_cases = sorted(source_regions["case_id"].astype(str).unique())
    if not expected_cases:
        raise ValueError("Source run contains no patients")
    if set(caches) != set(expected_cases):
        raise ValueError(
            f"Cache/case mismatch: caches={len(caches)}, region cases={len(expected_cases)}"
        )

def pca_tangents(points: np.ndarray) -> np.ndarray:
    points = np.asarray(points, float)
    overall = points[-1] - points[0]
    output = []
    for index in range(len(points)):
        if index == 0:
            local = points[:3]
        elif index == len(points) - 1:
            local = points[-3:]
        else:
            local = points[index - 1:index + 2]
        centered = local - local.mean(axis=0)
        _, _, vh = np.linalg.svd(centered, full_matrices=False)
        direction = vh[0]
        if direction @ overall < 0:
            direction = -direction
        output.append(direction / np.linalg.norm(direction))
    return np.asarray(output)


def icc_absolute_agreement(values_a: np.ndarray, values_b: np.ndarray) -> float:
    matrix = np.column_stack((values_a, values_b)).astype(float)
    n, k = matrix.shape
    if n < 3:
        return math.nan
    grand = matrix.mean()
    row_means = matrix.mean(axis=1)
    column_means = matrix.mean(axis=0)
    ms_rows = k * np.sum((row_means - grand) ** 2) / (n - 1)
    ms_columns = n * np.sum((column_means - grand) ** 2) / (k - 1)
    residual = matrix - row_means[:, None] - column_means[None, :] + grand
    ms_error = np.sum(residual ** 2) / ((n - 1) * (k - 1))
    denominator = (
        ms_rows + (k - 1) * ms_error
        + k * (ms_columns - ms_error) / n
    )
    return float((ms_rows - ms_error) / denominator) if abs(denominator) > 1e-12 else math.nan


def comparison_summary(table: pd.DataFrame, variant: str, stratum: str) -> dict[str, object]:
    own = table if stratum == "all" else table.loc[table["baseline_confidence"] == stratum]
    both = own.loc[
        own["baseline_retained"] & own[f"{variant}_retained"]
        & own["baseline_frequency_cpm"].notna()
        & own[f"{variant}_frequency_cpm"].notna()
    ].copy()
    difference = both[f"{variant}_frequency_cpm"] - both["baseline_frequency_cpm"]
    absolute = difference.abs()
    return {
        "variant": variant,
        "baseline_stratum": stratum,
        "regions_n": int(len(own)),
        "both_retained_frequency_n": int(len(both)),
        "retained_status_agreement": float(
            (own["baseline_retained"] == own[f"{variant}_retained"]).mean()
        ) if len(own) else math.nan,
        "three_level_confidence_agreement": float(
            (own["baseline_confidence"] == own[f"{variant}_confidence"]).mean()
        ) if len(own) else math.nan,
        "baseline_retained_to_excluded_n": int(
            (own["baseline_retained"] & ~own[f"{variant}_retained"]).sum()
        ),
        "baseline_excluded_to_retained_n": int(
            (~own["baseline_retained"] & own[f"{variant}_retained"]).sum()
        ),
        "frequency_icc_absolute_agreement": (

                both["baseline_frequency_cpm"].to_numpy(),

def write_csv(path: Path, rows: list[dict], fallback: list[str]) -> None:
    fields: list[str] = []
    for row in rows:
        for field in row:
            if field not in fields:
                fields.append(field)

    with path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, extrasaction="ignore")
        writer.writeheader()
        writer.writerows(rows)


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()

        nodes_sha256=np.asarray([metadata["nodes_sha256"]]),
    )
    os.replace(temporary, path)


def load_cache(
    path: Path, expected: dict | None = None
) -> tuple[dict, np.ndarray, np.ndarray, np.ndarray]:
    with np.load(path, allow_pickle=False) as archive:
        if expected is not None:
            actual = {
                "cache_format_version": int(archive["cache_format_version"][0]),
                "case_id": str(archive["case_id"][0]),
                "video_path": str(archive["video_path"][0]),
                "video_size": int(archive["video_size"][0]),
                "video_mtime_ns": int(archive["video_mtime_ns"][0]),
                "tracking_sha256": str(archive["tracking_sha256"][0]),
                "nodes_sha256": str(archive["nodes_sha256"][0]),
            }
            wanted = {"cache_format_version": CACHE_FORMAT_VERSION, **expected}
            differences = [
                key for key, value in wanted.items() if actual.get(key) != value
            ]
            if differences:
                raise ValueError(
                    "Trajectory cache input mismatch: " + ", ".join(differences)
                )
        tracked = {
            "time_s": archive["time_s"],
            "positions": archive["positions"],
            "raw_positions": archive["raw_positions"],
            "corrected": archive["corrected"],
            "valid": archive["target_valid"],
            "raw_valid": archive["raw_target_valid"],
            "pose_valid": archive["pose_valid"],
            "epoch": archive["epoch"],
            "poses": archive["poses"],
            "target_ncc": archive["target_ncc"],
            "target_fb": archive["target_fb"],
            "point_brightness": archive["point_brightness"],
            "global_brightness": archive["global_brightness"],
            "heldout_raw": archive["heldout_raw"],
            "heldout_corrected": archive["heldout_corrected"],
            "training_inliers": archive["training_inliers"],
            "heldout_count": archive["heldout_count"],
        }
        return (
            tracked,
            archive["initial_points"],
            archive["anterior_axes"],
            archive["posterior_axes"],
        )


def subset_coverage(points: np.ndarray, chosen: tuple[int, ...]) -> float:
    """Euclidean span retained from the first-frame regional point set."""
    def diameter(values: np.ndarray) -> float:
        if len(values) < 2:
            return 0.0
        delta = values[:, None, :] - values[None, :, :]
        return float(np.sqrt((delta * delta).sum(axis=2)).max())

    full = diameter(points)
    if full <= 1e-12:
        return 1.0
    return diameter(points[list(chosen)]) / full


def choose_fixed_point_segment(
    tracked: dict,
    initial: np.ndarray,
    columns: np.ndarray,
    min_point_fraction: float,
    min_span_fraction: float,
) -> dict | None:
    """Choose one unchanged point set and one continuous epoch-contained segment."""
    number = len(columns)
    minimum = max(2, int(math.ceil(number * min_point_fraction)))
    best: dict | None = None
    for size in range(minimum, number + 1):
        for local_choice in itertools.combinations(range(number), size):
            coverage = subset_coverage(initial[columns], local_choice)
            if coverage + 1e-12 < min_span_fraction:
                continue
            selected = columns[list(local_choice)]
            valid = (
                tracked["pose_valid"]
                & tracked["valid"][:, selected].all(axis=1)
                & np.isfinite(tracked["corrected"][:, selected]).all(axis=(1, 2))
            )
            for start, stop in base.contiguous_runs(valid, tracked["epoch"]):
                duration = float(
                    tracked["time_s"][stop - 1] - tracked["time_s"][start]
                )
                candidate = {
                    "start": start,
                    "stop": stop,
                    "columns": selected,
                    "local_choice": local_choice,
                    "duration_s": duration,
                    "point_count": size,
                    "point_fraction": size / number,
                    "span_fraction": coverage,
                }
                key = (duration, size, coverage, -start)
                if best is None or key > best["key"]:
                    candidate["key"] = key
                    best = candidate
    if best is not None:
        best.pop("key")
    return best


def periodic_trend_comparison(
    time_s: np.ndarray, signal: np.ndarray, frequency_cpm: float
) -> dict:
    """Compare the frozen periodic model with an explicit quadratic slow trend."""
    tt = np.asarray(time_s, float) - float(np.mean(time_s))
    y = np.asarray(signal, float)
    quadratic = np.column_stack((np.ones(len(tt)), tt, tt * tt))
    phase = 2.0 * np.pi * tt * frequency_cpm / 60.0
    periodic = np.column_stack(
        (np.ones(len(tt)), tt, np.sin(phase), np.cos(phase))
    )
    quadratic_residual = y - quadratic @ np.linalg.lstsq(quadratic, y, rcond=None)[0]
    periodic_residual = y - periodic @ np.linalg.lstsq(periodic, y, rcond=None)[0]
    quadratic_sse = float(quadratic_residual @ quadratic_residual)
    periodic_sse = float(periodic_residual @ periodic_residual)
    improvement = (
        (quadratic_sse - periodic_sse) / quadratic_sse
        if quadratic_sse > 1e-12 else math.nan
    )
    return {
        "quadratic_trend_sse": quadratic_sse,
        "periodic_model_sse": periodic_sse,
        "periodic_over_quadratic_improvement": improvement,
    }


def fit_candidate(time_s: np.ndarray, signal: np.ndarray) -> dict:
    fit = frozen.fit_unified_frequency(time_s, signal)
    if fit.get("status") != "valid":
        return fit
    comparison = periodic_trend_comparison(
        time_s, signal, float(fit["dominant_frequency_cpm"])
    )
    return {**fit, **comparison}


def usability(
    fit: dict,
    duration_s: float,
    min_duration_s: float,
    min_cycles: float,
    min_periodic_improvement: float,
) -> tuple[bool, str]:
    reasons: list[str] = []
    if fit.get("status") != "valid":
        reasons.append(str(fit.get("reason", "candidate_frequency_missing")))
    else:
        if duration_s + 1e-12 < min_duration_s:
            reasons.append("continuous_duration_below_rule")
        if float(fit.get("observed_cycles", 0.0)) + 1e-12 < min_cycles:
            reasons.append("candidate_cycle_coverage_below_rule")
        improvement = fit.get("periodic_over_quadratic_improvement", math.nan)
        if not np.isfinite(improvement) or improvement < min_periodic_improvement:
            reasons.append("periodic_model_not_better_than_quadratic_trend")
    return not reasons, "pass" if not reasons else ";".join(reasons)


def projected_signal(
    tracked: dict,
    initial: np.ndarray,
    axes: np.ndarray,
    columns: np.ndarray,
    start: int,
    stop: int,
) -> np.ndarray:
    displacement = tracked["corrected"][start:stop, columns] - initial[columns]
    point_signal = np.einsum("tki,ki->tk", displacement, axes[columns])
    return point_signal.mean(axis=1)


def safe_correlation(a: np.ndarray, b: np.ndarray) -> float:
    valid = np.isfinite(a) & np.isfinite(b)
    if valid.sum() < 20 or np.std(a[valid]) < 1e-12 or np.std(b[valid]) < 1e-12:
        return math.nan
    return float(np.corrcoef(a[valid], b[valid])[0, 1])


def fixed_frequency_component(
    time_s: np.ndarray, signal: np.ndarray, frequency_cpm: float
) -> dict:
    """Measure one nominated frequency by joint trend-plus-sinusoid regression."""
    time_s = np.asarray(time_s, float)
    signal = np.asarray(signal, float)
    valid = np.isfinite(time_s) & np.isfinite(signal)
    if valid.sum() < 20:
        return {"score": math.nan, "amplitude": math.nan}
    tt = time_s[valid] - float(np.mean(time_s[valid]))
    y = signal[valid]
    trend = np.column_stack((np.ones(len(tt)), tt))
    trend_fitted = trend @ np.linalg.lstsq(trend, y, rcond=None)[0]
    residual = y - trend_fitted
    energy = float(residual @ residual)
    if energy <= 1e-12:
        return {"score": math.nan, "amplitude": 0.0}
    phase = 2.0 * np.pi * tt * frequency_cpm / 60.0
    waves = np.column_stack((np.sin(phase), np.cos(phase)))
    full_design = np.column_stack((trend, waves))
    full_coefficients = np.linalg.lstsq(full_design, y, rcond=None)[0]
    coefficients = full_coefficients[-2:]
    periodic_valid = waves @ coefficients
    full_residual = y - full_design @ full_coefficients
    score = float(np.clip(1.0 - (full_residual @ full_residual) / energy, 0, 1))
    # Preserve original frame alignment.  Different control channels can have
    # different missing frames; compressing each independently makes a later
    # correlation compare the wrong frames or fail on unequal lengths.
    residual = np.full(len(signal), np.nan, float)
    periodic = np.full(len(signal), np.nan, float)
    residual[valid] = y - trend_fitted
    periodic[valid] = periodic_valid
    return {
        "score": score,
        "amplitude": float(np.linalg.norm(coefficients)),
        "residual": residual,
        "periodic": periodic,
        "valid": valid,
        "coefficient": complex(float(coefficients[0]), float(coefficients[1])),
    }


def background_signal(
    tracked: dict,
    initial: np.ndarray,
    axes: np.ndarray,
    columns: np.ndarray,
    start: int,
    stop: int,
) -> np.ndarray:
    values = np.full(stop - start, np.nan, float)
    for output_index, frame in enumerate(range(start, stop)):
        if not tracked["pose_valid"][frame]:
            continue
        inverse = np.linalg.inv(tracked["poses"][frame])
        mapped = initial[columns] @ inverse[:2, :2].T + inverse[:2, 2]
        displacement = mapped - initial[columns]
        values[output_index] = np.einsum(
            "ki,ki->k", displacement, axes[columns]
        ).mean()
    return values


def artifact_evidence(
    tracked: dict,
    initial: np.ndarray,
    axes: np.ndarray,
    selected: dict | None,
    fit: dict,
    usable: bool,
    deformation_supported: bool,
) -> dict:
    """Describe signal origin using controls from the identical point set and segment."""
    if selected is None or fit.get("status") != "valid":
        return {
            "evidence_class": "insufficient_evidence",
            "evidence_reason": "optimized_frequency_or_segment_missing",
        }
    start, stop = selected["start"], selected["stop"]
    columns = selected["columns"]
    time_s = tracked["time_s"][start:stop]
    frequency = float(fit["dominant_frequency_cpm"])
    displacement = tracked["corrected"][start:stop, columns] - initial[columns]
    point_signals = np.einsum("tki,ki->tk", displacement, axes[columns])
    tissue = point_signals.mean(axis=1)
    background = background_signal(
        tracked, initial, axes, columns, start, stop
    )
    local_brightness = np.nanmean(
        tracked["point_brightness"][start:stop, columns], axis=1
    )
    global_brightness = tracked["global_brightness"][start:stop]

    tissue_frequency = fixed_frequency_component(time_s, tissue, frequency)
    background_frequency = fixed_frequency_component(time_s, background, frequency)
    local_brightness_frequency = fixed_frequency_component(
        time_s, local_brightness, frequency
    )
    global_brightness_frequency = fixed_frequency_component(
        time_s, global_brightness, frequency
    )

    def periodic_corr(control: dict) -> float:
        if "periodic" not in tissue_frequency or "periodic" not in control:
            return math.nan
        return safe_correlation(tissue_frequency["periodic"], control["periodic"])

    point_components = [
        fixed_frequency_component(time_s, point_signals[:, index], frequency)
        for index in range(point_signals.shape[1])
    ]
    # Keep participation and phase agreement as separate quantities.  The
    # normalized inverse-participation ratio is 1/n when one of n points owns
    # all same-frequency amplitude, and 1 when all points contribute equally.
    coefficients = np.asarray([
        item.get("coefficient", complex(0.0, 0.0))
        if np.isfinite(abs(item.get("coefficient", complex(math.nan, math.nan))))
        else complex(0.0, 0.0)
        for item in point_components
    ], complex)
    amplitudes = np.abs(coefficients)
    amplitude_sum = float(amplitudes.sum())
    amplitude_square_sum = float(amplitudes @ amplitudes)
    point_participation_fraction = (
        amplitude_sum ** 2 / (len(point_components) * amplitude_square_sum)
        if amplitude_square_sum > 1e-24 else 0.0
    )
    point_phase_coherence = (
        float(abs(coefficients.sum()) / amplitude_sum)
        if amplitude_sum > 1e-12 else math.nan
    )
    tissue_range = float(np.ptp(tissue))
    background_range = float(np.ptp(background[np.isfinite(background)])) if np.isfinite(background).any() else math.nan
    background_ratio = (
        background_range / tissue_range if tissue_range > 1e-12 else math.nan
    )
    ncc = tracked["target_ncc"][start:stop, columns]
    fb = tracked["target_fb"][start:stop, columns]
    mean_ncc = float(np.nanmean(ncc)) if np.isfinite(ncc).any() else math.nan
    p90_fb = float(np.nanpercentile(fb, 90)) if np.isfinite(fb).any() else math.nan
    heldout = tracked["heldout_corrected"][start:stop]
    mean_heldout = (
        float(np.nanmean(heldout)) if np.isfinite(heldout).any() else math.nan
    )

    background_periodic_corr = periodic_corr(background_frequency)
    local_brightness_periodic_corr = periodic_corr(local_brightness_frequency)
    global_brightness_periodic_corr = periodic_corr(global_brightness_frequency)
    thresholds = EVIDENCE_THRESHOLDS
    background_flag = bool(
        np.isfinite(background_ratio)
        and background_ratio >= thresholds["background_to_tissue_range_ratio"]
        and background_frequency.get("score", math.nan)
        >= thresholds["control_fixed_frequency_score"]
        and abs(background_periodic_corr)
        >= thresholds["control_periodic_correlation"]
    )
    brightness_flag = bool(
        global_brightness_frequency.get("score", math.nan)
        >= thresholds["control_fixed_frequency_score"]
        and abs(global_brightness_periodic_corr)
        >= thresholds["control_periodic_correlation"]
    )
    tracking_flag = bool(
        point_participation_fraction < thresholds["minimum_point_participation_fraction"]
        or (np.isfinite(point_phase_coherence)
         and point_phase_coherence < thresholds["minimum_point_phase_coherence"])
        or (np.isfinite(mean_ncc) and mean_ncc < thresholds["minimum_mean_ncc"])
        or (np.isfinite(p90_fb)
            and p90_fb > thresholds["maximum_p90_forward_backward_px"])
    )
    artifact_reasons = []
    if tracking_flag:
        artifact_reasons.append("tracking_or_point_inconsistency")
    if artifact_reasons:
        evidence_class = "artifact_risk"
        evidence_reason = ";".join(artifact_reasons)
    elif background_flag and not deformation_supported:
        # A small periodic registration residual can remain after otherwise
        # successful rigid correction.  Without independent paired-wall
        # deformation, same-frequency background and local translation are not
        # identifiable, so do not retain the value as medium-confidence motion.
        evidence_class = "insufficient_evidence"
        evidence_reason = "background_same_frequency_without_deformation_support"
    elif (
        usable
        and point_participation_fraction >= thresholds["minimum_point_participation_fraction"]
        and np.isfinite(point_phase_coherence)
        and point_phase_coherence >= thresholds["supported_point_phase_coherence"]
    ):
        if deformation_supported:
            evidence_class = "local_tissue_motion_supported"
            evidence_reason = "usable_frequency;coherent_points;paired_deformation_support"
        else:
            evidence_class = "coordinated_tissue_displacement_supported"
            evidence_reason = "usable_frequency;participating_coherent_points;deformation_type_unresolved"
    else:
        evidence_class = "insufficient_evidence"
        evidence_reason = "local_deformation_support_incomplete"
    confounding_reasons = []
    if background_flag:
        confounding_reasons.append("background_same_frequency")
    if brightness_flag:
        confounding_reasons.append("global_brightness_same_frequency")
    return {
        "evidence_class": evidence_class,
        "evidence_reason": evidence_reason,
        "background_same_frequency_flag": background_flag,
        "brightness_same_frequency_flag": brightness_flag,
        "tracking_artifact_flag": tracking_flag,
        "same_frequency_confounding_flag": bool(confounding_reasons),
        "same_frequency_confounding_reason": ";".join(confounding_reasons),
        "tissue_fixed_frequency_score": tissue_frequency.get("score", ""),
        "background_fixed_frequency_score": background_frequency.get("score", ""),
        "background_periodic_correlation": background_periodic_corr,
        "background_to_tissue_range_ratio_same_segment": background_ratio,
        "local_brightness_fixed_frequency_score": local_brightness_frequency.get("score", ""),
        "local_brightness_periodic_correlation": local_brightness_periodic_corr,
        "global_brightness_fixed_frequency_score": global_brightness_frequency.get("score", ""),
        "global_brightness_periodic_correlation": global_brightness_periodic_corr,
        "point_phase_coherence": point_phase_coherence,
        "point_participation_fraction": point_participation_fraction,
        "mean_target_ncc_same_segment": mean_ncc,
        "p90_target_fb_px_same_segment": p90_fb,
        "mean_background_heldout_residual_px_same_segment": mean_heldout,
    }


def paired_deformation_support(
    tracked: dict,
    initial: np.ndarray,
    axes: np.ndarray,
    local_indices: np.ndarray,
    count: int,
    wall: str,
    selected: dict | None,
) -> dict:
    """Use the same selected anatomical pairs and the same continuous segment."""
    if selected is None:
        return {
            "status": "missing",
            "reason": "optimized_fixed_point_segment_missing",
        }
    chosen_local = local_indices[list(selected["local_choice"])]
    anterior = chosen_local
    posterior = chosen_local + count
    current = anterior if wall == "anterior" else posterior
    common_valid = (
        tracked["pose_valid"]
        & tracked["valid"][:, anterior].all(axis=1)
        & tracked["valid"][:, posterior].all(axis=1)
        & np.isfinite(tracked["corrected"][:, anterior]).all(axis=(1, 2))
        & np.isfinite(tracked["corrected"][:, posterior]).all(axis=(1, 2))
    )
    runs = base.contiguous_runs(common_valid, tracked["epoch"])
    if not runs:
        return {"status": "missing", "reason": "no_paired_common_valid_segment"}
    start, stop = max(
        runs,
        key=lambda run: tracked["time_s"][run[1] - 1] - tracked["time_s"][run[0]],
    )
    time_s = tracked["time_s"][start:stop]
    displacement = projected_signal(
        tracked, initial, axes, current, start, stop
    )
    initial_distance = np.linalg.norm(
        initial[posterior] - initial[anterior], axis=1
    )
    current_distance = np.linalg.norm(
        tracked["corrected"][start:stop, posterior]
        - tracked["corrected"][start:stop, anterior],
        axis=2,
    )
    strain = (current_distance / initial_distance[None] - 1.0).mean(axis=1)
    displacement_fit = fit_candidate(time_s, displacement)
    strain_fit = fit_candidate(time_s, strain)
    strain_range = float(np.ptp(strain))
    difference = (
        abs(
            float(displacement_fit["dominant_frequency_cpm"])
            - float(strain_fit["dominant_frequency_cpm"])
        )
        if displacement_fit.get("status") == strain_fit.get("status") == "valid"
        else math.nan
    )
    supported = bool(
        displacement_fit.get("status") == strain_fit.get("status") == "valid"
        and strain_range >= frozen.STRAIN_RANGE_THRESHOLD
        and difference <= frozen.FREQUENCY_TOLERANCE_CPM
    )
    reasons: list[str] = []
    if displacement_fit.get("status") != "valid":
        reasons.append("support_displacement_frequency_missing")
    if strain_fit.get("status") != "valid":
        reasons.append("paired_strain_frequency_missing")
    if strain_range < frozen.STRAIN_RANGE_THRESHOLD:
        reasons.append("strain_range_below_frozen_threshold")
    if np.isfinite(difference) and difference > frozen.FREQUENCY_TOLERANCE_CPM:
        reasons.append("frequency_difference_above_frozen_tolerance")
    return {
        "status": "valid",
        "reason": "pass" if supported else ";".join(reasons),
        "supported": supported,
        "duration_s": float(time_s[-1] - time_s[0]),
        "displacement_frequency_cpm": displacement_fit.get(
            "dominant_frequency_cpm", ""
        ),
        "strain_frequency_cpm": strain_fit.get("dominant_frequency_cpm", ""),
        "strain_range": strain_range,
        "frequency_difference_cpm": difference if np.isfinite(difference) else "",
        "paired_local_point_indices": ";".join(str(i) for i in chosen_local),
    }


def analyse_cache(
    case_id: str,
    cache_path: Path,
    args: argparse.Namespace,
    cache_identity: dict,
) -> list[dict]:
    tracked, initial, anterior_axes, posterior_axes = load_cache(
        cache_path, cache_identity
    )
    axes = np.vstack((anterior_axes, posterior_axes))
    count = len(anterior_axes)
    regions = dict(zip(base.frozen.REGION_NAMES, np.array_split(np.arange(count), 3)))
    rows: list[dict] = []
    for wall, offset in (("anterior", 0), ("posterior", count)):
        for region, local_indices in regions.items():
            columns = local_indices + offset
            arrays = base.region_arrays(tracked, initial, axes, local_indices, columns)
            original_valid = arrays["region_valid"] & np.isfinite(arrays["corrected_mean"])
            original_runs = base.contiguous_runs(original_valid, tracked["epoch"])
            original_segment = None
            if original_runs:
                start, stop = max(
                    original_runs,
                    key=lambda run: tracked["time_s"][run[1] - 1]
                    - tracked["time_s"][run[0]],
                )
                original_duration = float(
                    tracked["time_s"][stop - 1] - tracked["time_s"][start]
                )
                original_fit = fit_candidate(
                    tracked["time_s"][start:stop], arrays["corrected_mean"][start:stop]
                )
                original_segment = (start, stop)
            else:
                original_duration = 0.0
                original_fit = {
                    "status": "missing",
                    "reason": "no_continuous_technical_valid_segment",
                }

            selected = choose_fixed_point_segment(
                tracked,
                initial,
                columns,
                args.min_point_fraction,
                args.min_span_fraction,
            )
            if selected is None:
                optimized_duration = 0.0
                optimized_fit = {
                    "status": "missing",
                    "reason": "no_fixed_point_segment_with_required_coverage",
                }
            else:
                start, stop = selected["start"], selected["stop"]
                optimized_duration = selected["duration_s"]
                optimized_signal = projected_signal(
                    tracked,
                    initial,
                    axes,
                    selected["columns"],
                    start,
                    stop,
                )
                optimized_fit = fit_candidate(
                    tracked["time_s"][start:stop], optimized_signal
                )

            usable, reason = usability(
                optimized_fit,
                optimized_duration,
                args.min_duration_s,
                args.min_cycles,
                args.min_periodic_improvement,
            )
            deformation = paired_deformation_support(
                tracked,
                initial,
                axes,
                local_indices,
                count,
                wall,
                selected,
            )
            evidence = artifact_evidence(
                tracked,
                initial,
                axes,
                selected,
                optimized_fit,
                usable,
                bool(deformation.get("supported", False)),
            )
            row = {
                "case_id": case_id,
                "wall": wall,
                "region": region,
                "actual_video_duration_s": float(tracked["time_s"][-1]),
                "original_candidate_status": original_fit.get("status", "missing"),
                "original_candidate_reason": original_fit.get("reason", ""),
                "original_candidate_frequency_cpm": original_fit.get(
                    "dominant_frequency_cpm", ""
                ),
                "original_segment_duration_s": original_duration,
                "original_observed_cycles": original_fit.get("observed_cycles", ""),
                "original_boundary_hit": original_fit.get("search_boundary", ""),
                "optimized_candidate_status": optimized_fit.get("status", "missing"),
                "optimized_candidate_reason": optimized_fit.get("reason", ""),
                "optimized_candidate_frequency_cpm": optimized_fit.get(
                    "dominant_frequency_cpm", ""
                ),
                "optimized_segment_duration_s": optimized_duration,
                "optimized_observed_cycles": optimized_fit.get("observed_cycles", ""),
                "optimized_boundary_hit": optimized_fit.get("search_boundary", ""),
                "optimized_score": optimized_fit.get("score", ""),
                "quadratic_trend_sse": optimized_fit.get("quadratic_trend_sse", ""),

def patient_rows(case_ids: list[str], regions: list[dict], status: dict[str, dict]) -> list[dict]:
    grouped: dict[str, list[dict]] = defaultdict(list)
    for row in regions:
        grouped[row["case_id"]].append(row)
    output: list[dict] = []
    for case_id in case_ids:
        own = grouped[case_id]
        original = [row for row in own if row["original_candidate_status"] == "valid"]
        optimized = [row for row in own if row["frequency_usable"]]
        retained = [row for row in own if row.get("retained_for_patient_model")]
        durations = [
            float(row["optimized_segment_duration_s"])
            for row in own
            if row["frequency_usable"]
        ]
        output.append({
            "case_id": case_id,
            "processing_status": status[case_id]["status"],
            "processing_reason": status[case_id]["reason"],
            "original_candidate_region_count": len(original),
            "optimized_usable_region_count": len(optimized),
            "retained_region_count": len(retained),
            "retained_anterior_region_count": sum(
                row["wall"] == "anterior" for row in retained),
            "retained_posterior_region_count": sum(
                row["wall"] == "posterior" for row in retained),
            "optimized_usable_duration_median_s": (
                statistics.median(durations) if durations else ""
            ),
            "optimized_boundary_region_count": sum(
                bool(row["optimized_boundary_hit"])
                for row in own if row["frequency_usable"]
            ),
            "optimized_support_insufficient_region_count": sum(
                not bool(row["frequency_usable"]) for row in own
            ),
            "optimized_deformation_supported_region_count": sum(
                bool(row["optimized_deformation_supported"]) for row in own
            ),
            "local_tissue_motion_supported_region_count": sum(
                row.get("evidence_class") == "local_tissue_motion_supported"
                for row in own
            ),
            "coordinated_tissue_displacement_supported_region_count": sum(
                row.get("evidence_class") == "coordinated_tissue_displacement_supported"
                for row in own
            ),
            "artifact_risk_region_count": sum(
                row.get("evidence_class") == "artifact_risk" for row in own
            ),
            "insufficient_evidence_region_count": sum(
                row.get("evidence_class") == "insufficient_evidence" for row in own
            ),
        })
    return output


def sensitivity_rows(regions: list[dict]) -> list[dict]:
    rows: list[dict] = []
    case_count = len({row["case_id"] for row in regions})
    for duration, cycles, improvement in itertools.product(
        DURATION_CANDIDATES_S,
        CYCLE_CANDIDATES,
        TREND_IMPROVEMENT_CANDIDATES,
    ):
        usable = []
        by_case: dict[str, int] = defaultdict(int)
        for row in regions:
            fit = {
                "status": row["optimized_candidate_status"],
                "reason": row["optimized_candidate_reason"],
                "observed_cycles": row["optimized_observed_cycles"],
                "periodic_over_quadratic_improvement": row[
                    "periodic_over_quadratic_improvement"
                ],
            }
            ok, _ = usability(
                fit,
                float(row["optimized_segment_duration_s"]),
                duration,
                cycles,
                improvement,
            )
            if ok:
                usable.append(row)
                by_case[row["case_id"]] += 1
        rows.append({
            "min_duration_s": duration,
            "min_candidate_cycles": cycles,
            "min_periodic_over_quadratic_improvement": improvement,
            "usable_region_count": len(usable),
            "patients_with_at_least_1_usable_region": sum(v >= 1 for v in by_case.values()),
            "patients_with_at_least_2_usable_regions": sum(v >= 2 for v in by_case.values()),
            "patients_total": case_count,
        })
    return rows

def perturbed_source(source: pd.Series, trim_front: float, trim_back: float) -> pd.Series:
    output = source.copy()
    start = int(longitudinal.finite_float(source.get("selected_start_frame"), -1))
    stop = int(longitudinal.finite_float(source.get("selected_stop_frame_exclusive"), -1))
    if start < 0 or stop <= start:
        return output
    length = stop - start
    output["selected_start_frame"] = start + int(math.floor(length * trim_front))
    output["selected_stop_frame_exclusive"] = stop - int(math.floor(length * trim_back))
    return output

def write_csv_rows(path: Path, rows: list[dict], fieldnames: list[str]) -> None:
    """Write a stable header even when every requested case failed."""
    with path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)


def union_fieldnames(rows: list[dict], fallback: list[str]) -> list[str]:
    fields = []
    for row in rows:
        for field in row:
            if field not in fields:
                fields.append(field)
    return fields or fallback


def read_case_manifest(path: Path) -> list[dict]:
    required = {"case_id", "video", "tracking", "nodes"}
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        missing = required - set(reader.fieldnames or ())
        if missing:
            raise ValueError(
                "Case manifest is missing columns: " + ", ".join(sorted(missing))
            )
        rows = list(reader)
    if not rows:
        raise ValueError("Case manifest contains no cases")
    case_ids = [row["case_id"].strip() for row in rows]
    if any(not case_id for case_id in case_ids):
        raise ValueError("Every manifest row must have a non-empty case_id")
    if len(set(case_ids)) != len(case_ids):
        raise ValueError("Case manifest contains duplicate case_id values")
    base_dir = path.resolve().parent
    resolved = []
    for source, case_id in zip(rows, case_ids):
        row = dict(source)
        row["case_id"] = case_id
        for field in ("video", "tracking", "nodes"):
            value = Path(row[field].strip())
            row[field] = value if value.is_absolute() else base_dir / value
        resolved.append(row)
    return resolved


def prepare_case(source: dict) -> dict:
    case_id = source["case_id"]
    for field in ("video", "tracking", "nodes"):
        if not source[field].is_file():
            raise FileNotFoundError(f"{case_id}: {field} not found: {source[field]}")
    frame_count, fps, width, height = base.video_metadata(source["video"])
    if fps <= 0 or frame_count < 2:
        raise ValueError(f"{case_id}: invalid video timing")
    if (width, height) != (1260, 910):
        raise ValueError(
            f"{case_id}: expected 1260x910 source video, found {width}x{height}"
        )
    with np.load(source["tracking"], allow_pickle=False) as archive:
        tracks = archive["tracks"]
        if tracks.ndim != 3 or tracks.shape[2] != 2:
            raise ValueError(f"{case_id}: tracking array must have shape (T, N, 2)")
        tracking_frames = int(tracks.shape[0])
    if tracking_frames < 1:
        raise ValueError(f"{case_id}: tracking file contains no first-frame seed")
    spacing_text = source.get("mm_per_processed_pixel", "").strip()
    spacing = math.nan
    if spacing_text:
        spacing = float(spacing_text)
        if not math.isfinite(spacing) or spacing <= 0:
            raise ValueError(f"{case_id}: mm_per_processed_pixel must be positive")
    node_rows = json.loads(source["nodes"].read_text(encoding="utf-8-sig"))
    anterior = [row for row in node_rows if row.get("rail_name") == "anterior_wall"]
    posterior = [row for row in node_rows if row.get("rail_name") == "posterior_wall"]
    if len(anterior) != len(posterior) or len(anterior) < 6:
        raise ValueError(f"{case_id}: unmatched or insufficient anterior/posterior nodes")
    return dict(
        anonymous_id=case_id,
        holdout_id=case_id,
        key=hashlib.sha256(case_id.encode("utf-8")).hexdigest(),
        video=source["video"],
        tracking=source["tracking"],
        nodes=source["nodes"],
        wall_points=len(anterior),
        spacing=spacing,
        frame_count=frame_count,
        tracking_frames=tracking_frames,
        fps=fps,
        available_s=(frame_count - 1) / fps,
        frame_count_match=frame_count == tracking_frames,
    )


def robust_background_view(gray: np.ndarray, fan_mask: np.ndarray) -> np.ndarray:
    """Remove framewise affine brightness drift inside the ultrasound fan."""
    valid = fan_mask & (gray > 12)
    if int(valid.sum()) < 100:
        return gray.copy()
    q25, median, q75 = np.percentile(gray[valid], [25.0, 50.0, 75.0])
    scale = max(float(q75 - q25), 1.0)
    output = np.zeros_like(gray)
    values = 128.0 + 48.0 * (gray.astype(float) - float(median)) / scale
    output[valid] = np.clip(values[valid], 16.0, 240.0).astype(np.uint8)
    return output


def track_normalized_background(row: dict, duration_s: float) -> tuple:
    gray0, initial, anterior_axes, posterior_axes = base.frozen.load_background(row)
    fps = float(row["fps"])
    frame_limit = int(math.floor(duration_s * fps + 1e-9)) + 1
    time_s = np.arange(frame_limit, dtype=float) / fps
    n, count = frame_limit, len(initial)
    positions = np.full((n, count, 2), np.nan, np.float32)
    raw_positions = np.full_like(positions, np.nan)
    valid = np.zeros((n, count), bool)
    raw_valid = np.zeros_like(valid)
    corrected = np.full((n, count, 2), np.nan, float)
    pose_valid = np.zeros(n, bool)
    epoch = np.zeros(n, int)
    poses = np.full((n, 3, 3), np.nan, float)
    target_ncc = np.full((n, count), np.nan, float)
    target_fb = np.full((n, count), np.nan, float)
    point_brightness = np.full((n, count), np.nan, float)
    global_brightness = np.full(n, np.nan, float)
    heldout_raw = np.full(n, np.nan, float)
    heldout_corrected = np.full(n, np.nan, float)
    training_inliers = np.zeros(n, int)
    heldout_count = np.zeros(n, int)
    accumulator = np.eye(3)
    current_epoch = 0
    fan_mask = gray0 > 12
    cap = cv2.VideoCapture(str(row["video"]))
    previous = processed_previous = background_previous = None
    try:
        for index in range(n):
            ok, frame = cap.read()
            if not ok:
                raise ValueError(f"{row['anonymous_id']} missing frame {index}")
            gray = cv2.cvtColor(
                cv2.resize(frame, None, fx=0.5, fy=0.5, interpolation=cv2.INTER_AREA),
                cv2.COLOR_BGR2GRAY,
            )
            if gray.shape != (455, 630):
                raise ValueError("Frozen background ROI requires 630x455 processed frames")
            processed = base.local_contrast(gray)
            background = robust_background_view(gray, fan_mask)
            stats = {}
            if index == 0:
                positions[0] = raw_positions[0] = initial
                valid[0] = raw_valid[0] = True
                corrected[0] = initial
                pose_valid[0] = True
                poses[0] = accumulator
            else:
                live = np.flatnonzero(valid[index - 1])
                if len(live):
                    found, good, scores, fb = base.track_step(
                        processed_previous, processed, positions[index - 1, live])
                    target_ncc[index, live] = scores
                    target_fb[index, live] = fb
                    kept = live[good]
                    positions[index, kept] = found[good]
                    valid[index, kept] = True
                raw_live = np.flatnonzero(raw_valid[index - 1])
                if len(raw_live):
                    found, good, _, _ = base.track_step(
                        previous, gray, raw_positions[index - 1, raw_live])
                    kept = raw_live[good]
                    raw_positions[index, kept] = found[good]
                    raw_valid[index, kept] = True
                cv2.setRNGSeed(RANDOM_SEED)
                matrix, stats = base.rigid_step(
                    background_previous, background, raw_positions[index - 1, raw_live]
                ) if len(raw_live) >= 3 else (None, {})
                if matrix is None:
                    accumulator = np.eye(3)
                    current_epoch += 1
                else:
                    accumulator = accumulator @ np.linalg.inv(
                        np.vstack((matrix, [0.0, 0.0, 1.0])))
                    pose_valid[index] = True
                    poses[index] = accumulator
                    corrected[index, valid[index]] = (
                        positions[index, valid[index]] @ accumulator[:2, :2].T
                        + accumulator[:2, 2]
                    )
            epoch[index] = current_epoch
            for column in np.flatnonzero(valid[index]):
                point_brightness[index, column] = base.patch_median(
                    gray, positions[index, column])
            global_brightness[index] = float(np.median(gray[fan_mask]))
            heldout_raw[index] = stats.get("heldout_raw_median_px", math.nan)
            heldout_corrected[index] = stats.get(
                "heldout_corrected_median_px", math.nan)
            training_inliers[index] = stats.get("training_inliers", 0)
            heldout_count[index] = stats.get("heldout_count", 0)
            previous = gray
            processed_previous = processed
            background_previous = background
            if index and index % 300 == 0:
                print(row["anonymous_id"], "B", index, n, flush=True)
    finally:
        cap.release()
    tracked = dict(
        time_s=time_s, positions=positions, raw_positions=raw_positions,
        valid=valid, raw_valid=raw_valid, corrected=corrected,
        pose_valid=pose_valid, epoch=epoch, poses=poses,
        target_ncc=target_ncc, target_fb=target_fb,
        point_brightness=point_brightness,
        global_brightness=global_brightness,
        heldout_raw=heldout_raw, heldout_corrected=heldout_corrected,
        training_inliers=training_inliers, heldout_count=heldout_count,
    )
    return tracked, initial, anterior_axes, posterior_axes


def correlation(a: np.ndarray, b: np.ndarray) -> float:
    good = np.isfinite(a) & np.isfinite(b)
    if int(good.sum()) < 20 or np.std(a[good]) < 1e-9 or np.std(b[good]) < 1e-9:
        return math.nan
    return float(np.corrcoef(a[good], b[good])[0, 1])


def fit_unified_frequency(time_s: np.ndarray, signal: np.ndarray) -> dict:
    if len(time_s) < 20:
        return dict(status="missing", reason="technical_segment_too_short")
    tt = time_s - time_s.mean()
    trend_x = np.column_stack((np.ones(len(tt)), tt))
    trend = trend_x @ np.linalg.lstsq(trend_x, signal, rcond=None)[0]
    residual = signal - trend
    energy = float(residual @ residual)
    if energy / len(signal) < 1e-12:
        return dict(status="missing", reason="technical_nearly_constant")
    scores = []
    for frequency in GRID_CPM:
        phase = 2.0 * np.pi * tt * frequency / 60.0
        design = np.column_stack((trend_x, np.sin(phase), np.cos(phase)))
        fitted = design @ np.linalg.lstsq(design, signal, rcond=None)[0]
        scores.append(np.clip(1.0 - float(((signal - fitted) ** 2).sum()) / energy, 0.0, 1.0))
    scores = np.asarray(scores)
    best_index = int(np.argmax(scores))
    best = float(GRID_CPM[best_index])
    period_s = 60.0 / best
    lag = np.interp(time_s + period_s, time_s, residual, left=np.nan, right=np.nan)
    half = np.interp(time_s + period_s / 2.0, time_s, residual, left=np.nan, right=np.nan)
    return dict(
        status="valid", reason="unified_fixed_band_dominant_frequency",
        dominant_frequency_cpm=best, period_s=period_s,
        score=float(scores[best_index]),
        one_period_correlation=correlation(residual, lag),
        half_period_correlation=correlation(residual, half),
        search_boundary=bool(best_index in (0, len(GRID_CPM) - 1)),
        observed_cycles=float(best * (time_s[-1] - time_s[0]) / 60.0),
    )


def analyse_branch(
    case_id: str,
    branch: str,
    tracked: dict,
    initial: np.ndarray,
    anterior_axes: np.ndarray,
    posterior_axes: np.ndarray,
    spacing: float,
) -> list[dict]:
    all_axes = np.vstack((anterior_axes, posterior_axes))
    count = len(anterior_axes)
    regions = dict(zip(base.frozen.REGION_NAMES, np.array_split(np.arange(count), 3)))
    rows = []
    for wall, offset in (("anterior", 0), ("posterior", count)):
        for region, local_indices in regions.items():
            columns = local_indices + offset
            arrays = base.region_arrays(tracked, initial, all_axes, local_indices, columns)
            raw_valid = (
                arrays["region_valid"]
                & np.isfinite(arrays["corrected_mean"])
            )
            raw_runs = base.contiguous_runs(raw_valid, tracked["epoch"])
            if not raw_runs:
                rows.append(dict(
                    case_id=case_id, branch=branch, wall=wall, region=region,
                    status="missing", reason="no_continuous_technical_valid_segment",
                    dominant_frequency_cpm="", actual_video_duration_s=tracked["time_s"][-1],
                    continuous_valid_duration_s=0.0,
                    paired_wall_strain_range="",
                    support_displacement_frequency_status="missing",
                    support_displacement_frequency_cpm="",
                    paired_wall_strain_frequency_cpm="",
                    paired_wall_continuous_valid_duration_s=0.0,
                    support_common_valid_duration_s=0.0,
                    displacement_strain_frequency_difference_cpm="",
                    deformation_supported=False,
                    deformation_support_reason="raw_frequency_missing",
                    mean_heldout_residual_px="",
                    background_to_raw_range_ratio="",
                ))
                continue
            start, stop = max(
                raw_runs,
                key=lambda own: tracked["time_s"][own[1] - 1] - tracked["time_s"][own[0]],
            )
            own = slice(start, stop)
            fit = fit_unified_frequency(
                tracked["time_s"][own], arrays["corrected_mean"][own])
            support_common_valid = (
                raw_valid
                & arrays["pair_valid"]
                & np.isfinite(arrays["radial_strain"])
            )
            support_runs = base.contiguous_runs(
                support_common_valid, tracked["epoch"]
            )
            if support_runs:
                support_start, support_stop = max(
                    support_runs,
                    key=lambda run: (
                        tracked["time_s"][run[1] - 1] - tracked["time_s"][run[0]]
                    ),
                )
                support_own = slice(support_start, support_stop)
                support_displacement_fit = fit_unified_frequency(
                    tracked["time_s"][support_own],
                    arrays["corrected_mean"][support_own],
                )
                strain_fit = fit_unified_frequency(
                    tracked["time_s"][support_own],
                    arrays["radial_strain"][support_own],
                )
                strain_range = float(np.ptp(arrays["radial_strain"][support_own]))
                support_duration = float(
                    tracked["time_s"][support_stop - 1]
                    - tracked["time_s"][support_start]
                )
            else:
                support_displacement_fit = dict(
                    status="missing", reason="no_common_support_valid_segment"
                )
                strain_fit = dict(status="missing", reason="no_paired_wall_valid_segment")
                strain_range = math.nan
                support_duration = 0.0
            frequency_difference = (
                abs(float(support_displacement_fit["dominant_frequency_cpm"])
                    - float(strain_fit["dominant_frequency_cpm"]))
                if support_displacement_fit["status"] == strain_fit["status"] == "valid"
                else math.nan
            )
            deformation_supported = bool(
                fit["status"] == "valid"
                and support_displacement_fit["status"] == strain_fit["status"] == "valid"
                and strain_range >= STRAIN_RANGE_THRESHOLD
                and frequency_difference <= FREQUENCY_TOLERANCE_CPM
            )
            support_reasons = []
            if fit["status"] != "valid":
                support_reasons.append("raw_frequency_missing")
            if support_displacement_fit["status"] != "valid":
                support_reasons.append("support_displacement_frequency_missing")
            if strain_fit["status"] != "valid":
                support_reasons.append("strain_frequency_missing")
            if not np.isfinite(strain_range):
                support_reasons.append("paired_wall_segment_missing")
            elif strain_range < STRAIN_RANGE_THRESHOLD:
                support_reasons.append("strain_range_below_0p0045")
            if (np.isfinite(frequency_difference)
                    and frequency_difference > FREQUENCY_TOLERANCE_CPM):
                support_reasons.append("frequency_difference_above_0p25")
            metrics = base.segment_metrics(case_id, wall, region, tracked, arrays)
            primary = max(metrics, key=lambda item: item["duration_s"]) if metrics else {}
            row = dict(
                case_id=case_id, branch=branch, wall=wall, region=region,
                actual_video_duration_s=float(tracked["time_s"][-1]),
                continuous_valid_duration_s=float(
                    tracked["time_s"][stop - 1] - tracked["time_s"][start]),
                valid_frames=stop - start,
                frequency_range_cpm="0.5-5.0",
                range_mm=(
                    float(np.ptp(arrays["corrected_mean"][own]) * spacing)
                    if math.isfinite(spacing) else ""
                ),
                pose_valid_fraction=float(tracked["pose_valid"].mean()),
                epoch_count=int(tracked["epoch"].max() + 1),
                local_brightness_fractional_range=primary.get(
                    "local_brightness_fractional_range", ""),
                radial_strain_range=primary.get("radial_strain_range", ""),
                paired_wall_strain_range=(
                    strain_range if np.isfinite(strain_range) else ""
                ),
                support_displacement_frequency_status=(
                    support_displacement_fit["status"]
                ),
                support_displacement_frequency_cpm=(
                    support_displacement_fit.get("dominant_frequency_cpm", "")
                ),
                paired_wall_continuous_valid_duration_s=support_duration,
                support_common_valid_duration_s=support_duration,
                paired_wall_strain_frequency_status=strain_fit["status"],
                paired_wall_strain_frequency_cpm=strain_fit.get(
                    "dominant_frequency_cpm", ""),
                displacement_strain_frequency_difference_cpm=(
                    frequency_difference if np.isfinite(frequency_difference) else ""),
                deformation_supported=deformation_supported,
                deformation_support_reason=(
                    "pass" if deformation_supported else ";".join(support_reasons)),
                background_to_raw_range_ratio=primary.get(
                    "background_to_raw_range_ratio", ""),
                corrected_local_brightness_corr=primary.get(
                    "corrected_local_brightness_corr", ""),
                adjacent_point_corr=primary.get("adjacent_point_corr", ""),
                mean_heldout_residual_px=primary.get(
                    "mean_heldout_residual_px", ""),
            )
            row.update(fit)
            row.setdefault("dominant_frequency_cpm", "")
            rows.append(row)
    return rows

def run(args: argparse.Namespace) -> int:
    if args.out.exists():
        raise ValueError("Output exists; choose a new directory")
    args.out.mkdir(parents=True)
    manifest_rows = read_case_manifest(args.case_manifest)
    selected = []
    region_rows = []
    case_status_rows = []
    logs = [
        "rule=full_video_fixed_0.5_to_5.0_cpm_all_technical_valid_regions",
        "branch_A=raw_gray_background_registration",
        "branch_B=median_IQR_normalized_background_registration",
        "target_tracking_unchanged=true",
        "final_curve_smoothing=false",
        "deformation_support_comparison=longest_same_common_valid_segment",
        "duration_source=full_video_frame_count",
        "scientific_core_parameters_modified=false",
    ]
    started = time.time()
    for source in manifest_rows:
        case_id = source["case_id"]
        try:
            row = prepare_case(source)
            selected.append(row)
        except Exception as exc:
            case_status_rows.append(dict(
                case_id=case_id, status="failed_input", reason=str(exc),
                actual_video_duration_s="", frame_count_match="",
                formal_A_status="not_run", formal_A_reason="",
                diagnostic_B_status="not_run", diagnostic_B_reason="",
            ))
            logs.append(f"{case_id}_failed_input={type(exc).__name__}:{exc}")
            continue
        duration_s = float(row["available_s"])
        case_started = time.time()
        try:
            print(row["anonymous_id"], "A start", flush=True)
            tracked_a, initial_a, anterior_a, posterior_a = base.track_real_case(
                row, duration_s)
            a_region_rows = analyse_branch(
